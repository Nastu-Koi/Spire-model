"""Offline coverage/depth audit of retained dev frames; never executes the engine.

Run from repository root:
    python3 .scratch/public-search-repair/research/audit_current.py
"""

import hashlib
import json
import sys
from collections import Counter
from pathlib import Path

ROOT = Path(__file__).resolve().parents[3]
sys.path.insert(0, str(ROOT))

from combat_solver_cli.public_effects import CombatModel, content
from combat_solver_cli.public_search import PublicPlanner, public_view


def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def main():
    base = ROOT / "combat_solver_cli/artifacts/repair-11/evaluation-final"
    report = json.loads((base / "report.json").read_text())
    counts, reasons, blockers, coverage, ties = (Counter() for _ in range(5))
    cases, inputs = [], {}
    for run in report["runs"]:
        if run["profile"] != "fixed" or run["repetition"] != 1:
            continue
        path = base / run["run_dir"] / "attempts/000001/decisions.jsonl"
        inputs[str(path.relative_to(ROOT))] = digest(path)
        local = Counter()
        planners = {
            d: PublicPlanner(depth=d, beam_width=16, sampling_seed=run["sampling_seed"])
            for d in (1, 3, 6)
        }
        for line in path.read_text().splitlines():
            row = json.loads(line)
            view = public_view(row["frame"])
            decisions = {d: p.choose(row["frame"]) for d, p in planners.items()}
            counts["frames"] += 1
            local["frames"] += 1
            counts["baseline_score_mismatches"] += (
                decisions[3][1]["candidate_scores"]
                != row["diagnostics"]["candidate_scores"]
            )
            if view["public"]["phase"] != "combat":
                scores = [s["score"] for s in decisions[3][1]["candidate_scores"]]
                if len(scores) > 1 and len(set(scores)) == 1:
                    ties[view["public"]["phase"]] += 1
                continue
            counts["combat_frames"] += 1
            local["combat_frames"] += 1
            model = CombatModel(view)
            counts["unsupported_hook_frames"] += model.unsupported_hooks
            for lo, hi in ((1, 3), (3, 6)):
                changed = decisions[lo][0] != decisions[hi][0]
                key = f"depth_{lo}_to_{hi}_action_changes"
                counts[key] += changed
                local[key] += changed
                if model.unsupported_hooks:
                    counts[key + "_with_unsupported_hooks"] += changed
            for depth, (_, diag) in decisions.items():
                counts[f"depth_{depth}_expanded"] += diag["expanded"]
            for entity in model.entities:
                kind, ident = entity.get("entity_type"), content(entity)
                if (
                    kind == "power"
                    and ident.removesuffix("_POWER") not in model.safe_powers
                ) or (kind == "relic" and ident not in model.safe_relics):
                    blockers[f"{kind}:{ident}"] += 1
            for candidate in view["legal"]["candidates"]:
                if candidate["verb"] in {"PLAY_CARD", "USE_POTION"}:
                    prediction = model.predict(candidate)
                    reasons[prediction.reason or "modeled"] += 1
        effects = base / run["run_dir"] / "effects.jsonl"
        inputs[str(effects.relative_to(ROOT))] = digest(effects)
        for line in effects.read_text().splitlines():
            coverage[json.loads(line)["prediction_coverage"]] += 1
        cases.append({"seed": run["seed"], **local})
    result = {
        "scope": "offline replay of four dev seeds, fixed profile, repetition 1",
        "limitations": [
            "Depth variants only rescore baseline observations, not their own trajectories.",
            "Action changes do not establish better decisions or higher win rate.",
            "Blocker occurrence counts overlap and do not attribute deaths.",
            "No holdout, native engine, or wall-clock throughput experiment was run.",
        ],
        "counts": counts,
        "root_play_or_potion_reasons": reasons,
        "blocker_frame_occurrences": dict(blockers.most_common()),
        "executed_effect_coverage": coverage,
        "outside_all_equal_multiple_candidates": ties,
        "cases": cases,
        "source_sha256": {
            str(p.relative_to(ROOT)): digest(p)
            for p in [
                ROOT / "combat_solver_cli/public_search.py",
                ROOT / "combat_solver_cli/public_effects.py",
                ROOT / "combat_solver_cli/astar.py",
                Path(__file__).resolve(),
            ]
        },
        "input_sha256": inputs,
    }
    target = Path(__file__).with_name("current-audit.json")
    target.write_text(json.dumps(result, ensure_ascii=False, indent=2) + "\n")
    print(json.dumps({"counts": counts, "root_reasons": reasons}, ensure_ascii=False))
    assert counts["baseline_score_mismatches"] == 0, "Frozen baseline scores changed"


if __name__ == "__main__":
    main()
